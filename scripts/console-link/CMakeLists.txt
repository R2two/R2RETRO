cmake_minimum_required(VERSION 3.16)
project(R2N64ConsoleLinkCheck LANGUAGES C CXX)
if(NOT DEFINED R2N64_CONSOLE_LIBDIR)
    message(FATAL_ERROR "Set R2N64_CONSOLE_LIBDIR to the isolated lib directory")
endif()
add_executable(console_link_check main.cpp)
if(PS4)
    target_link_libraries(console_link_check PRIVATE "--whole-archive")
else()
    target_link_libraries(console_link_check PRIVATE "-Wl,--whole-archive")
endif()
target_link_libraries(console_link_check PRIVATE
    "${R2N64_CONSOLE_LIBDIR}/libfceumm_libretro.a"
    "${R2N64_CONSOLE_LIBDIR}/libbsnes_mercury_libretro.a")
if(PS4)
    target_link_libraries(console_link_check PRIVATE "--no-whole-archive" SceNet)
else()
    target_link_libraries(console_link_check PRIVATE "-Wl,--no-whole-archive" m dl)
    add_executable(fceumm_palette_check palette.cpp)
    set_property(TARGET fceumm_palette_check PROPERTY CXX_STANDARD 17)
    target_link_libraries(fceumm_palette_check PRIVATE "${R2N64_CONSOLE_LIBDIR}/libfceumm_libretro.a" m)
    target_link_options(fceumm_palette_check PRIVATE "-Wl,--wrap=fceumm_filestream_read")
    add_executable(bsnes_chip_check snes-chips.cpp)
    set_property(TARGET bsnes_chip_check PROPERTY CXX_STANDARD 17)
    target_link_libraries(bsnes_chip_check PRIVATE "${R2N64_CONSOLE_LIBDIR}/libbsnes_mercury_libretro.a" m dl)
endif()
