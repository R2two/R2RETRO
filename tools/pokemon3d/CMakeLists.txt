cmake_minimum_required(VERSION 3.16)
project(R2N64Pokemon3DLab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(PkgConfig REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
pkg_check_modules(SDL REQUIRED IMPORTED_TARGET sdl2 SDL2_image)
add_library(pokemon3d_scene scene.cpp)
add_library(pokemon3d_red_state red_state.cpp)
target_compile_options(pokemon3d_red_state PRIVATE -Wall -Wextra -Wpedantic)
target_compile_options(pokemon3d_scene PRIVATE -Wall -Wextra -Wpedantic)
add_library(pokemon3d_renderer renderer.cpp)
target_link_libraries(pokemon3d_renderer PUBLIC pokemon3d_scene PkgConfig::SDL)
target_compile_options(pokemon3d_renderer PRIVATE -Wall -Wextra -Wpedantic)
add_executable(pokemon3d main.cpp)
target_link_libraries(pokemon3d PRIVATE pokemon3d_renderer)
target_compile_options(pokemon3d PRIVATE -Wall -Wextra -Wpedantic)
add_executable(pokemon3d_tests tests.cpp)
target_link_libraries(pokemon3d_tests PRIVATE pokemon3d_renderer)
target_compile_options(pokemon3d_tests PRIVATE -Wall -Wextra -Wpedantic)
enable_testing()
add_library(pokemon3d_view_settings view_settings.cpp)
target_compile_options(pokemon3d_view_settings PRIVATE -Wall -Wextra -Wpedantic)
add_executable(pokemon3d_view_settings_tests view_settings_tests.cpp)
target_link_libraries(pokemon3d_view_settings_tests PRIVATE pokemon3d_view_settings)
add_test(NAME view_preference COMMAND pokemon3d_view_settings_tests)
add_executable(pokemon3d_red_state_tests red_state_tests.cpp)
target_link_libraries(pokemon3d_red_state_tests PRIVATE pokemon3d_red_state)
add_test(NAME red_live_state_synthetic COMMAND pokemon3d_red_state_tests)
if(TARGET r2n64_emulation)
    add_executable(pokemon3d_live live.cpp)
    target_link_libraries(pokemon3d_live PRIVATE pokemon3d_view_settings pokemon3d_red_state pokemon3d_renderer r2n64_emulation)
    target_compile_options(pokemon3d_live PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_definitions(pokemon3d_live PRIVATE R2N64_LAB_ROOT="${CMAKE_SOURCE_DIR}")
endif()
add_test(NAME map_decoder_synthetic COMMAND ${Python3_EXECUTABLE}
    "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/pokemon_red_map_tests.py")
add_test(NAME map_mesh_synthetic COMMAND ${Python3_EXECUTABLE}
    "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/pokemon_red_mesh_tests.py")
add_test(NAME textured_world_synthetic COMMAND ${Python3_EXECUTABLE}
    "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/pokemon_red_world_tests.py")
add_test(NAME scene_validation COMMAND pokemon3d_tests --validation "${CMAKE_BINARY_DIR}/evidence")
add_test(NAME depth_and_orbit COMMAND pokemon3d_tests --render "${CMAKE_BINARY_DIR}/evidence")
set_tests_properties(depth_and_orbit PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=offscreen;LIBGL_ALWAYS_SOFTWARE=1;MESA_GLES_VERSION_OVERRIDE=2.0" TIMEOUT 30)
